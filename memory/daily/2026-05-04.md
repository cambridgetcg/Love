**Utility signature** (place in `x/knowledge/keeper/utils.go`):

```go
// StakeWeightedCardinality computes the fraction of totalStake held by the
// deduplicated set of addresses, expressed in [0, 1_000_000].
// Primitive for stake-weighted quorum (#25) and domain diversity (#27).
func StakeWeightedCardinality(
    addresses []string,
    stakeGetter func(addr string) uint64,
    totalStake uint64,
) uint64 {
    if totalStake == 0 {
        return 0
    }
    seen := make(map[string]struct{}, len(addresses))
    var weightedSum uint64
    for _, addr := range addresses {
        if _, dup := seen[addr]; dup {
            continue
        }
        seen[addr] = struct{}{}
        weightedSum += stakeGetter(addr)
    }
    return safeMulDiv(weightedSum, 1_000_000, totalStake)
}
```

```go
// EconomicInvariantViolation describes a single invariant failure.
type EconomicInvariantViolation struct {
    Invariant   string `json:"invariant"`
    Expected    string `json:"expected"`
    Actual      string `json:"actual"`
    Description string `json:"description"`
}

// ValidateEconomicInvariants checks all economic integrity invariants.
// Returns an empty slice if all invariants hold.
func (k Keeper) ValidateEconomicInvariants(ctx sdk.Context) []EconomicInvariantViolation {
    var violations []EconomicInvariantViolation
    stats := k.GetMeterStats(ctx)

    minted, _ := sdkmath.NewIntFromString(stats.TotalMinted)
    routed, _ := sdkmath.NewIntFromString(stats.TotalRouted)
    burned, _ := sdkmath.NewIntFromString(stats.TotalBurned)
    locked, _ := sdkmath.NewIntFromString(stats.TotalLocked)

    // Invariant 1: Cannot route more than minted
    if routed.GT(minted) {
        violations = append(violations, EconomicInvariantViolation{
            Invariant:   "routed_leq_minted",
            Expected:    "TotalRouted <= TotalMinted",
            Actual:      fmt.Sprintf("routed=%s minted=%s", routed, minted),
            Description: "More tokens routed to research than minted — impossible without external injection",
        })
    }

    // Invariant 2: Cannot burn more than minted
    if burned.GT(minted) {
        violations = append(violations, EconomicInvariantViolation{
            Invariant:   "burned_leq_minted",
            Expected:    "TotalBurned <= TotalMinted",
            Actual:      fmt.Sprintf("burned=%s minted=%s", burned, minted),
            Description: "More tokens burned than minted — impossible supply state",
        })
    }

    // Invariant 3: Circulating supply <= bank module supply
    // circulating = minted - locked - burned (approximate; bank is authoritative)
    bankSupply := k.bankKeeper.GetSupply(ctx, "ulgm").Amount
    computedCirculating := minted.Sub(locked).Sub(burned)
    if computedCirculating.GT(bankSupply) {
        violations = append(violations, EconomicInvariantViolation{
            Invariant:   "circulating_leq_bank_supply",
            Expected:    "TotalMinted - TotalLocked - TotalBurned <= BankSupply",
            Actual:      fmt.Sprintf("computed=%s bank=%s", computedCirculating, bankSupply),
            Description: "Computed circulating supply exceeds bank module supply — meter undercount (likely flag #31/#32)",
        })
    }

    return violations
}
```
